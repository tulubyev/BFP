🐘 PostgreSQL + PostGIS установка

docker-compose.yml (PostGIS)
version: '3.8'

services:
  database:
    image: postgis/postgis:15-3.3
    container_name: baikal-postgis
    restart: unless-stopped
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    environment:
      - POSTGRES_PASSWORD=${DB_PASSWORD}
      - POSTGRES_USER=${DB_USER}
      - POSTGRES_DB=baikal_gis
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${DB_USER}"]
      interval: 30s
      timeout: 10s
      retries: 3

volumes:
  postgres_data:

Примечание: Образ postgis/postgis включает все необходимые расширения, включая PostGIS, pgRouting и другие геопространственные расширения.

version: '3.8'

services:
  titiler:
    image: ghcr.io/developmentseed/titiler:latest
    container_name: baikal-titiler
    restart: unless-stopped
    ports:
      - "8000:8000"
    environment:
      - GDAL_DISABLE_READDIR_ON_OPEN=EMPTY_DIR
      - CPL_VSIL_CURL_ALLOWED_EXTENSIONS=.tif,.TIF,.nc
      - AWS_REQUEST_PAYER=requester
      - DEFAULT_MAX_SIZE=8192
    depends_on:
      - redis

  redis:
    image: redis:alpine
    container_name: baikal-redis
    restart: unless-stopped
    ports:
      - "6379:6379"

Особенности настройки: Переменные GDAL оптимизируют работу с COG файлами в облаке, уменьшая количество HTTP запросов.

const TITILER_URL = 'https://tiles.baikal-gis.ru';

// Динамический URL для NDVI тайлов
function getNDVITileUrl(cogUrl) {
  const params = new URLSearchParams({
    url: cogUrl,
    expression: '(B08-B04)/(B08+B04)',
    colormap_name: 'rdylgn',
    rescale: '-0.2,0.8'
  });
  
  return `${TITILER_URL}/cog/tiles/{z}/{x}/{y}?${params}`;
}

// Добавление слоя на MapLibre
map.addSource('ndvi-layer', {
  type: 'raster',
  tiles: [getNDVITileUrl(selectedCOG)],
  tileSize: 256
});

  id: 'ndvi',
  type: 'raster',
  source: 'ndvi-layer',
  paint: { 'raster-opacity': 0.8 }

🔍 STAC Catalog Search

backend/stac_search.py
from pystac_client import Client
from shapely.geometry import box

# Baikal region bounding box
BAIKAL_BBOX = [103.5, 51.4, 110.5, 56.0]

def search_sentinel2(start_date, end_date, max_cloud=20):
    """Поиск снимков Sentinel-2 для Байкала"""
    
    client = Client.open(
        "https://earth-search.aws.element84.com/v1"
    )
    
    search = client.search(
        collections=["sentinel-2-l2a"],
        bbox=BAIKAL_BBOX,
        datetime=f"{start_date}/{end_date}",
        query={
            "eo:cloud_cover": {"lt": max_cloud}
        },
        sortby=[{"field": "datetime", "direction": "desc"}]
    )
    
    items = list(search.items())
    return [{
        "id": item.id,
        "datetime": item.datetime,
        "cloud_cover": item.properties["eo:cloud_cover"],
        "cog_url": item.assets["visual"].href
    } for item in items]

🐘 PostGIS Spatial Query

queries/forest_loss_analysis.sql
-- Анализ убыли лесов по районам БПТ
WITH forest_changes AS (
  SELECT
    r.name AS region_name,
    ST_Area(
      ST_Intersection(fc.geom, r.geom)::geography
    ) / 10000 AS loss_ha,
    fc.loss_year,
    fc.loss_type  -- fire, logging, disease
  FROM forest_cover_changes fc
  JOIN bpt_regions r 
    ON ST_Intersects(fc.geom, r.geom)
  WHERE fc.loss_year BETWEEN 2015 AND 2024
)
SELECT
  region_name,
  loss_year,
  loss_type,
  ROUND(SUM(loss_ha)::numeric, 2) AS total_loss_ha,
  COUNT(*) AS patch_count
FROM forest_changes
GROUP BY region_name, loss_year, loss_type
ORDER BY loss_year DESC, total_loss_ha DESC;

-- Создание GIST индекса для ускорения
CREATE INDEX idx_forest_geom 
  ON forest_cover_changes 
  USING GIST (geom);

🔌 Интеграция MapLibre + TiTiler

src/components/BaikalMap.jsx (React)
import 'maplibre-gl/dist/maplibre-gl.css';
import { useEffect, useRef } from 'react';
import maplibregl from 'maplibre-gl';

const BaikalMap = () => {
  const mapContainer = useRef(null);
  const map = useRef(null);

  useEffect(() => {
    if (map.current) return; // Инициализировать карту только один раз
    
    map.current = new maplibregl.Map({
      container: mapContainer.current,
      style: {
        version: 8,
        sources: {},
        layers: []
      },
      center: [105.5, 53.0], // Центр Байкала
      zoom: 7
    });

    // Добавление базового слоя OSM
    map.current.on('load', () => {
      map.current.addSource('osm', {
        type: 'raster',
        tiles: [
          'https://a.tile.openstreetmap.org/{z}/{x}/{y}.png',
          'https://b.tile.openstreetmap.org/{z}/{x}/{y}.png'
        ],
        tileSize: 256
      });

      map.current.addLayer({
        id: 'osm',
        type: 'raster',
        source: 'osm',
        minzoom: 0,
        maxzoom: 19
      });

      // Добавление слоя NDVI из TiTiler
      map.current.addSource('ndvi', {
        type: 'raster',
        tiles: [
          'https://your-titiler-endpoint.example/cog/tiles/{z}/{x}/{y}?url=s3://your-bucket/path/to/cog.tif&expression=(B08-B04)/(B08+B04)&colormap_name=rdylgn'
        ],
        tileSize: 256
      });

      map.current.addLayer({
        id: 'ndvi-layer',
        type: 'raster',
        source: 'ndvi',
        paint: {
          'raster-opacity': 0.8
        }
      });
    });
  }, []);

  return (
    <div className="map-wrapper">
      <div ref={mapContainer} className="map-container" />
    </div>
  );
};

export default BaikalMap;
