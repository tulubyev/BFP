sql — init-scripts/01-init-postgis.sql 
-- Включение расширений PostGIS
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS postgis_topology;
CREATE EXTENSION IF NOT EXISTS postgis_raster;
CREATE EXTENSION IF NOT EXISTS pg_trgm;  -- для полнотекстового поиска

-- Создание схемы для ГИС данных
CREATE SCHEMA IF NOT EXISTS gis;

-- Таблица лесничеств
CREATE TABLE gis.forest_districts (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    region VARCHAR(100),
    area_ha NUMERIC(12,2),
    geom GEOMETRY(MultiPolygon, 4326),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Таблица событий (пожары, вырубки)
CREATE TABLE gis.forest_events (
    id SERIAL PRIMARY KEY,
    event_type VARCHAR(50) NOT NULL,  -- 'fire', 'logging', 'disease'
    detected_date DATE,
    area_ha NUMERIC(10,2),
    severity VARCHAR(20),
    satellite_source VARCHAR(50),
    geom GEOMETRY(Polygon, 4326),
    metadata JSONB
);

-- Пространственные индексы (GIST)
CREATE INDEX idx_districts_geom ON gis.forest_districts USING GIST (geom);
CREATE INDEX idx_events_geom ON gis.forest_events USING GIST (geom);
CREATE INDEX idx_events_date ON gis.forest_events (detected_date);

-- Проверка установки PostGIS
SELECT PostGIS_Version();

Nginx reverse proxy

Маршрутизация запросов и SSL

nginx/nginx.conf 
events {
    worker_connections 1024;
}

http {
    # Кэширование тайлов
    proxy_cache_path /var/cache/nginx/tiles levels=1:2 
        keys_zone=tiles_cache:100m max_size=10g inactive=7d;
    
    # Gzip сжатие
    gzip on;
    gzip_types application/json application/geo+json image/png;
    
    upstream api_backend {
        server api:8000;
    }
    
    upstream titiler_backend {
        server titiler:8000;
    }
    
    upstream frontend_backend {
        server frontend:3000;
    }

    server {
        listen 80;
        server_name baikal-gis.ru;
        
        # Редирект на HTTPS
        location / {
            return 301 https://$server_name$request_uri;
        }
        
        # Let's Encrypt challenge
        location /.well-known/acme-challenge/ {
            root /var/www/certbot;
        }
    }

    server {
        listen 443 ssl http2;
        server_name baikal-gis.ru;
        
        ssl_certificate /etc/nginx/ssl/fullchain.pem;
        ssl_certificate_key /etc/nginx/ssl/privkey.pem;
        
        # Frontend (SPA)
        location / {
            proxy_pass http://frontend_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
        }
        
        # REST API
        location /api/ {
            proxy_pass http://api_backend/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
        
        # TiTiler - тайловый сервер с кэшированием
        location /tiles/ {
            proxy_pass http://titiler_backend/;
            proxy_set_header Host $host;
            
            # Кэширование тайлов на 7 дней
            proxy_cache tiles_cache;
            proxy_cache_valid 200 7d;
            proxy_cache_key $scheme$request_uri;
            add_header X-Cache-Status $upstream_cache_status;
            
            # CORS для MapLibre
            add_header Access-Control-Allow-Origin *;
        }
        
        # WebSocket для real-time алертов
        location /ws/ {
            proxy_pass http://api_backend/ws/;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }
    }
}

// Конфигурация
const CONFIG = {
    TITILER_URL: '/tiles',  // через nginx proxy
    API_URL: '/api',
    BAIKAL_CENTER: [107.5, 53.5],
    DEFAULT_ZOOM: 7
};

// Инициализация карты с OpenStreetMap подложкой
const map = new maplibregl.Map({
    container: 'map',
    style: {
        version: 8,
        sources: {
            // OpenStreetMap тайлы (бесплатно, без API ключа)
            'osm': {
                type: 'raster',
                tiles: [
                    'https://tile.openstreetmap.org/{z}/{x}/{y}.png'
                ],
                tileSize: 256,
                attribution: '© OpenStreetMap contributors'
            }
        },
        layers: [{
            id: 'osm-layer',
            type: 'raster',
            source: 'osm'
        }]
    },
    center: CONFIG.BAIKAL_CENTER,
    zoom: CONFIG.DEFAULT_ZOOM
});

// Добавление слоя спутниковых данных из TiTiler
function addSatelliteLayer(cogUrl, layerId, options = {}) {
    const params = new URLSearchParams({
        url: cogUrl,
        bidx: options.bands || '4,3,2',  // RGB по умолчанию
        rescale: options.rescale || '0,3000'
    });
    
    map.addSource(layerId, {
        type: 'raster',
        tiles: [
            `${CONFIG.TITILER_URL}/cog/tiles/{z}/{x}/{y}.png?${params}`
        ],
        tileSize: 256
    });
    
    map.addLayer({
        id: layerId,
        type: 'raster',
        source: layerId,
        paint: { 'raster-opacity': options.opacity || 0.8 }
    });
}

// Добавление NDVI слоя с цветовой шкалой
function addNDVILayer(cogUrl) {
    const params = new URLSearchParams({
        url: cogUrl,
        expression: '(b8-b4)/(b8+b4)',  // NDVI формула
        colormap_name: 'rdylgn',
        rescale: '-0.2,0.8'
    });
    
    map.addSource('ndvi', {
        type: 'raster',
        tiles: [`${CONFIG.TITILER_URL}/cog/tiles/{z}/{x}/{y}.png?${params}`]
    });
    
    map.addLayer({
        id: 'ndvi-layer',
        type: 'raster',
        source: 'ndvi'
    });
}

// Загрузка GeoJSON событий (пожары, вырубки) из API
async function loadForestEvents() {
    const response = await fetch(`${CONFIG.API_URL}/events/geojson`);
    const geojson = await response.json();
    
    map.addSource('forest-events', {
        type: 'geojson',
        data: geojson
    });
    
    // Пожары - красные полигоны
    map.addLayer({
        id: 'fires',
        type: 'fill',
        source: 'forest-events',
        filter: ['==', 'event_type', 'fire'],
        paint: {
            'fill-color': '#ef4444',
            'fill-opacity': 0.6
        }
    });
}

// Инициализация при загрузке карты
map.on('load', () => {
    loadForestEvents();
    
    // Пример COG файла из Yandex S3
    const cogUrl = 'https://storage.yandexcloud.net/baikal-cog/sentinel2/2024/T48UXA_20240115.tif';
    addSatelliteLayer(cogUrl, 'sentinel-rgb');
});
