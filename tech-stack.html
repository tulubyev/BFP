GeoJSON API Routes

REST endpoints с типизацией

src/routes/events.ts 
import { FastifyPluginAsync } from 'fastify';
import { AppDataSource } from '../database';
import { ForestEvent } from '../entities/ForestEvent';

interface EventsQuery {
  eventType?: string;
  year?: number;
  bbox?: string; // "minX,minY,maxX,maxY"
}

export const eventsRoutes: FastifyPluginAsync = async (app) => {
  const repo = AppDataSource.getRepository(ForestEvent);

  app.get<{ Querystring: EventsQuery }>('/geojson', async (req) => {
    const { eventType, year, bbox } = req.query;
    
    let qb = repo
      .createQueryBuilder('e')
      .select([
        'e.id', 'e.eventType', 'e.areaHa',
        'ST_AsGeoJSON(e.geom)::json AS geometry'
      ]);

    if (eventType) {
      qb = qb.andWhere('e.eventType = :eventType', { eventType });
    }
    if (year) {
      qb = qb.andWhere('EXTRACT(YEAR FROM e.detectedDate) = :year', { year });
    }
    if (bbox) {
      const [minX, minY, maxX, maxY] = bbox.split(',').map(Number);
      qb = qb.andWhere(
        'ST_Intersects(e.geom, ST_MakeEnvelope(:minX, :minY, :maxX, :maxY, 4326))',
        { minX, minY, maxX, maxY }
      );
    }

    const events = await qb.getRawMany();
    
    return {
      type: 'FeatureCollection',
      features: events.map(e => ({
        type: 'Feature',
        geometry: e.geometry,
        properties: { id: e.id, eventType: e.eventType, areaHa: e.areaHa }
      }))
    };
  });
};

  region: 'ru-central1',
  endpoint: 'https://storage.yandexcloud.net',
  credentials: {
    accessKeyId: process.env.YC_ACCESS_KEY!,

export async function uploadCOG(
  key: string, 
  body: Buffer
): Promise<string> {
  await s3Client.send(new PutObjectCommand({
    Bucket: BUCKET,
    Key: key,
    Body: body,
    ContentType: 'image/tiff'
  }));
  return `https://storage.yandexcloud.net/${BUCKET}/${key}`;
}

export async function getPresignedUrl(
  key: string, 
  expiresIn = 3600
): Promise<string> {
  const command = new GetObjectCommand({ Bucket: BUCKET, Key: key });
  return getSignedUrl(s3Client, command, { expiresIn });
}

TiTiler Proxy Service

Интеграция с тайл-сервером

src/services/titiler.ts 
import axios, { AxiosInstance } from 'axios';

interface TileParams {
  url: string;
  expression?: string;  // e.g., "(b8-b4)/(b8+b4)" для NDVI
  colormap?: string;
  rescale?: string;
}

interface COGInfo {
  bounds: [number, number, number, number];
  minzoom: number;
  maxzoom: number;
  band_descriptions: string[][];
}

export class TiTilerService {
  private client: AxiosInstance;

  constructor(baseURL = 'http://titiler:8000') {
    this.client = axios.create({ baseURL });
  }

  // Получить URL для тайлов
  getTileUrl(params: TileParams): string {
    const qs = new URLSearchParams();
    qs.set('url', params.url);
    if (params.expression) qs.set('expression', params.expression);
    if (params.colormap) qs.set('colormap_name', params.colormap);
    if (params.rescale) qs.set('rescale', params.rescale);
    
    return `/cog/tiles/{z}/{x}/{y}.png?${qs}`;
  }

  // Получить метаданные COG
  async getCOGInfo(cogUrl: string): Promise<COGInfo> {
    const { data } = await this.client.get('/cog/info', {
      params: { url: cogUrl }
    });
    return data;
  }

  // NDVI preset
  getNDVITileUrl(cogUrl: string): string {
    return this.getTileUrl({
      url: cogUrl,
      expression: '(b8-b4)/(b8+b4)',
      colormap: 'rdylgn',
      rescale: '-0.2,0.8'
    });
  }
}

src/queues/imageProcessor.ts 
import { Queue, Worker, Job } from 'bullmq';
import { Redis } from 'ioredis';

const connection = new Redis(process.env.REDIS_URL!);

interface ProcessImageJob {
  sceneId: string;
  cogUrl: string;
  analysisType: 'ndvi' | 'nbr' | 'classification';
}

// Очередь задач
export const imageQueue = new Queue<ProcessImageJob>('image-processing', {
  connection,
  defaultJobOptions: {
    attempts: 3,
    backoff: { type: 'exponential', delay: 5000 }
  }
});

// Воркер обработки
export const imageWorker = new Worker<ProcessImageJob>(
  'image-processing',
  async (job: Job<ProcessImageJob>) => {
    const { sceneId, cogUrl, analysisType } = job.data;
    
    job.log(`Processing ${sceneId} with ${analysisType}`);
    
    // 1. Скачать COG и рассчитать индекс
    const result = await calculateIndex(cogUrl, analysisType);
    
    // 2. Найти аномалии (изменения)
    const changes = await detectChanges(result);
    
    // 3. Сохранить в PostGIS
    await saveToDatabase(changes);
    
    return { changesFound: changes.length };
  },
  { connection, concurrency: 2 }
);

from rio_cogeo.cogeo import cog_translate
from rio_cogeo.profiles import cog_profiles

def convert_to_cog(input_path, output_path):
    """Конвертация GeoTIFF в Cloud Optimized GeoTIFF"""
    
    output_profile = cog_profiles.get("deflate")
    output_profile.update({
        "blockxsize": 512,
        "blockysize": 512,
        "BIGTIFF": "IF_SAFER"
    })
    
    cog_translate(
        input_path,
        output_path,
        output_profile,
        overview_level=5,  # Пирамиды
        overview_resampling="average"
    )

📊 Схема базы данных

SQL — Полная схема для PostGIS 
-- Схема для ГИС данных
CREATE SCHEMA IF NOT EXISTS gis;

-- Расширения
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS postgis_raster;

-- Таблица регионов БПТ
CREATE TABLE gis.regions (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    region_type VARCHAR(50),  -- oblast, republic, krai
    area_ha NUMERIC(12,2),
    forest_cover_pct NUMERIC(5,2),
    geom GEOMETRY(MultiPolygon, 4326)
);

-- Таблица лесничеств
CREATE TABLE gis.forest_districts (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    region_id INTEGER REFERENCES gis.regions(id),
    area_ha NUMERIC(12,2),
    conifer_pct NUMERIC(5,2),
    geom GEOMETRY(MultiPolygon, 4326)
);

-- Таблица событий (пожары, вырубки, болезни)
CREATE TABLE gis.forest_events (
    id SERIAL PRIMARY KEY,
    event_type VARCHAR(50) NOT NULL,
    severity VARCHAR(20),  -- low, medium, high, critical
    detected_date DATE NOT NULL,
    confirmed_date DATE,
    area_ha NUMERIC(10,2),
    satellite_source VARCHAR(50),  -- sentinel2, landsat8
    scene_id VARCHAR(100),
    confidence NUMERIC(5,2),  -- 0-100%
    geom GEOMETRY(Polygon, 4326),
    metadata JSONB DEFAULT '{}'
);

-- Таблица временных рядов NDVI
CREATE TABLE gis.ndvi_timeseries (
    id SERIAL PRIMARY KEY,
    district_id INTEGER REFERENCES gis.forest_districts(id),
    observation_date DATE NOT NULL,
    mean_ndvi NUMERIC(6,4),
    min_ndvi NUMERIC(6,4),
    max_ndvi NUMERIC(6,4),
    cloud_cover_pct NUMERIC(5,2)
);

-- Пространственные индексы GIST
CREATE INDEX idx_regions_geom ON gis.regions USING GIST (geom);
CREATE INDEX idx_districts_geom ON gis.forest_districts USING GIST (geom);
CREATE INDEX idx_events_geom ON gis.forest_events USING GIST (geom);

-- Индексы для фильтрации
CREATE INDEX idx_events_type_date ON gis.forest_events (event_type, detected_date);
CREATE INDEX idx_ndvi_district_date ON gis.ndvi_timeseries (district_id, observation_date);

🐘 TypeORM Data Source

src/database.ts 
import { DataSource } from 'typeorm';
import { SnakeNamingStrategy } from 'typeorm-naming-strategies';
import { ForestEvent } from './entities/ForestEvent';
import { ForestDistrict } from './entities/ForestDistrict';
import { NdviTimeseries } from './entities/NdviTimeseries';

export const AppDataSource = new DataSource({
  type: 'postgres',
  
  // Подключение через переменные окружения
  host: process.env.DB_HOST || 'localhost',
  port: parseInt(process.env.DB_PORT || '5432'),
  username: process.env.DB_USER || 'gis_admin',
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME || 'baikal_gis',
  
  // Схема для ГИС данных
  schema: 'gis',
  
  // Entities
  entities: [ForestEvent, ForestDistrict, NdviTimeseries],
  
  // Миграции
  migrations: ['src/migrations/*.ts'],
  
  // snake_case для PostgreSQL
  namingStrategy: new SnakeNamingStrategy(),
  
  // Логирование SQL в dev режиме
  logging: process.env.NODE_ENV === 'development',
  
  // Синхронизация схемы (только для dev!)
  synchronize: false
});

// Инициализация подключения
export async function initDatabase(): Promise<void> {
  await AppDataSource.initialize();
  console.log('📦 Database connected');
  
  // Проверка PostGIS
  const result = await AppDataSource.query(
    'SELECT PostGIS_Version()'
  );
  console.log(`🌍 PostGIS: ${result[0].postgis_version}`);
}
