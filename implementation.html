Интеграция с TiTiler

src/components/BaikalMap.jsx
import 'maplibre-gl/dist/maplibre-gl.css';
import { useEffect, useRef } from 'react';
import maplibregl from 'maplibre-gl';

const BaikalMap = () => {
  const mapContainer = useRef(null);
  const map = useRef(null);

  useEffect(() => {
    if (map.current) return;
    
    map.current = new maplibregl.Map({
      container: mapContainer.current,
      style: {
        version: 8,
        sources: {},
        layers: []
      },
      center: [105.5, 53.0],
      zoom: 7
    });

    map.current.on('load', () => {
      // Добавление слоя NDVI из TiTiler
      map.current.addSource('ndvi', {
        type: 'raster',
        tiles: [
          'https://your-titiler-endpoint/cog/tiles/{z}/{x}/{y}?url=s3://bucket/path/to/cog.tif&expression=(B08-B04)/(B08+B04)&colormap_name=rdylgn'
        ],
        tileSize: 256
      });

      map.current.addLayer({
        id: 'ndvi-layer',
        type: 'raster',
        source: 'ndvi',
        paint: {
          'raster-opacity': 0.8
        }
      });
    });
  }, []);

  return (
    <div ref={mapContainer} className="w-full h-full" />
  );
};

export default BaikalMap;

Детали конфигурации

☁️ Конфигурация Yandex S3

1. Создание бакета

  1. 1. Войдите в консоль Yandex Cloud
  2. 2. Перейдите в Object Storage
  3. 3. Создайте бакет "baikal-forest-gis"
  4. 4. Установите ACL: "Публичный доступ к чтению"

2. Настройка credentials

~/.aws/credentials
[default]
aws_access_key_id = YOUR_ACCESS_KEY
aws_secret_access_key = YOUR_SECRET_KEY

3. Переменные окружения для TiTiler

.env
AWS_S3_ENDPOINT_URL=https://storage.yandexcloud.net
AWS_ACCESS_KEY_ID=YOUR_ACCESS_KEY
AWS_SECRET_ACCESS_KEY=YOUR_SECRET_KEY
AWS_REGION=ru-central1

🐘 PostgreSQL + PostGIS

Добавление PostGIS расширения

PostGIS уже включен в образ postgis/postgis, но можно проверить:

SQL
-- Подключение к базе данных
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS postgis_topology;
-- Проверка установки
SELECT PostGIS_Version();

Пример таблицы лесов

SQL
CREATE TABLE forest_areas (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    area_ha DECIMAL(12, 2),
    forest_type VARCHAR(100),
    geom GEOMETRY(POLYGON, 4326),
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_forest_geom ON forest_areas USING GIST (geom);

Оптимизация для больших данных

  • • Используйте GIST индексы для геометрий
  • • Разбивайте большие таблицы по годам/регионам
  • • Настройте параметры shared_buffers и work_mem
